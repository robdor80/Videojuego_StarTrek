cmake_minimum_required(VERSION 3.25)
project(TrekF04 LANGUAGES CXX)
set(CORERPG_SOURCE_DIR "" CACHE PATH "Exact Core worktree")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/core-revision.txt" EXPECTED_CORE_SHA)
string(STRIP "${EXPECTED_CORE_SHA}" EXPECTED_CORE_SHA)
execute_process(COMMAND git -C "${CORERPG_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE CORE_SHA OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT CORE_SHA STREQUAL EXPECTED_CORE_SHA)
 message(FATAL_ERROR "Core revision mismatch: ${CORE_SHA}; expected ${EXPECTED_CORE_SHA}")
endif()
execute_process(COMMAND git -C "${CORERPG_SOURCE_DIR}" diff --exit-code HEAD RESULT_VARIABLE DIRTY OUTPUT_QUIET)
if(NOT DIRTY EQUAL 0)
 message(FATAL_ERROR "Core tracked sources dirty")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_catalog.py" "${CMAKE_CURRENT_BINARY_DIR}/f04_catalog.h" COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_TESTING OFF CACHE BOOL "Core suite executed separately" FORCE)
set(CORERPG_BUILD_CONTENT_JSON OFF CACHE BOOL "Existing IDs projected into in-memory catalog" FORCE)
add_subdirectory("${CORERPG_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/corerpg")
set_target_properties(CoreRPG PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(TrekF04Bridge SHARED src/f04_bridge.cpp)
target_link_libraries(TrekF04Bridge PRIVATE CoreRPG::CoreRPG)
target_include_directories(TrekF04Bridge PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(TrekF04Bridge PRIVATE cxx_std_20)
target_compile_definitions(TrekF04Bridge PRIVATE F04_BRIDGE_BUILD F04_CORE_SHA="${CORE_SHA}")
if(MSVC)
 target_compile_options(TrekF04Bridge PRIVATE /W4 /permissive- /EHsc /utf-8)
endif()
