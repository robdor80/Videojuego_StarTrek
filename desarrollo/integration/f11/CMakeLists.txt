cmake_minimum_required(VERSION 3.25)
project(TrekF11 LANGUAGES CXX)
set(F11_ENABLE_PROMOTIONS ON CACHE BOOL "Explicit F11 Trek extension" FORCE)
add_subdirectory(../f05 "${CMAKE_CURRENT_BINARY_DIR}/f05")
enable_testing()
# A separate test binary contains the private, compiled native trial issuer.
# The product bridge exports no bootstrap, grant, evaluation or approval API.
add_library(TrekF11TrialHost SHARED ../f04/src/f04_bridge.cpp)
target_link_libraries(TrekF11TrialHost PRIVATE CoreRPG::CoreRPG TrekF05Language)
target_compile_features(TrekF11TrialHost PRIVATE cxx_std_20)
target_include_directories(TrekF11TrialHost PUBLIC include ../f04/include ../f05/include
 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/f05/f04" ../f05/src src)
file(READ ../f05/core-revision.txt F11_CORE_SHA)
string(STRIP "${F11_CORE_SHA}" F11_CORE_SHA)
target_compile_definitions(TrekF11TrialHost PRIVATE F04_BRIDGE_BUILD F04_CORE_SHA="${F11_CORE_SHA}"
 F05_ENABLE_ORDERS=1 F04_HISTORY_CORE=1 F04_HISTORY_TESTING=1
 F11_ENABLE_PROMOTIONS=1 F11_NATIVE_TEST_HOST=1)
if(MSVC)
 target_compile_options(TrekF11TrialHost PRIVATE /utf-8)
endif()
add_executable(f11_tests tests/f11_tests.cpp)
target_link_libraries(f11_tests PRIVATE TrekF11TrialHost CoreRPG::CoreRPG)
target_compile_features(f11_tests PRIVATE cxx_std_20)
target_compile_definitions(f11_tests PRIVATE F11_LEGACY_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../f04/tests/fixtures")
foreach(case IN ITEMS promotion negatives authority identity idempotency atomic corrupt legacy history abi)
 add_test(NAME f11.${case} COMMAND f11_tests ${case} "${CMAKE_CURRENT_BINARY_DIR}/${case}")
endforeach()
add_test(NAME f11.process_save COMMAND f11_tests process_save "${CMAKE_CURRENT_BINARY_DIR}/process")
set_tests_properties(f11.process_save PROPERTIES FIXTURES_SETUP f11_process)
add_test(NAME f11.process_load COMMAND f11_tests process_load "${CMAKE_CURRENT_BINARY_DIR}/process")
set_tests_properties(f11.process_load PROPERTIES FIXTURES_REQUIRED f11_process)
add_executable(f11_product_tests tests/f11_product_tests.cpp)
target_link_libraries(f11_product_tests PRIVATE TrekF04Bridge CoreRPG::CoreRPG ${CMAKE_DL_LIBS})
target_compile_features(f11_product_tests PRIVATE cxx_std_20)
set_target_properties(f11_product_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/f05/f04")
add_test(NAME f11.product_bridge COMMAND f11_product_tests "${CMAKE_CURRENT_BINARY_DIR}/process")
set_tests_properties(f11.product_bridge PROPERTIES FIXTURES_REQUIRED f11_process)
