cmake_minimum_required(VERSION 3.25)
project(TrekF02 LANGUAGES CXX)
set(CORERPG_SOURCE_DIR "" CACHE PATH "Unmodified CoreRPG checkout")
if(NOT EXISTS "${CORERPG_SOURCE_DIR}/include/corerpg/world_state.hpp")
  message(FATAL_ERROR "Set CORERPG_SOURCE_DIR to the real CoreRPG checkout")
endif()
execute_process(COMMAND git -C "${CORERPG_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE CORE_SHA OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT CORE_SHA STREQUAL "4f282863737bcd8f5add401202a9e1064c133672")
  message(FATAL_ERROR "Unaudited CoreRPG revision: ${CORE_SHA}. Reaudit before updating the pin.")
endif()
execute_process(COMMAND git -C "${CORERPG_SOURCE_DIR}" diff --exit-code HEAD
  RESULT_VARIABLE CORE_DIRTY OUTPUT_QUIET)
if(NOT CORE_DIRTY EQUAL 0)
  message(FATAL_ERROR "CoreRPG tracked sources differ from the audited revision")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Core is tested separately" FORCE)
set(CORERPG_BUILD_CONTENT_JSON OFF CACHE BOOL "Not required by this identity lab" FORCE)
add_subdirectory("${CORERPG_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/corerpg")
set_target_properties(CoreRPG PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(TrekF02Bridge SHARED src/f02_bridge.cpp)
target_link_libraries(TrekF02Bridge PRIVATE CoreRPG::CoreRPG)
target_include_directories(TrekF02Bridge PUBLIC include)
target_compile_features(TrekF02Bridge PRIVATE cxx_std_20)
target_compile_definitions(TrekF02Bridge PRIVATE F02_BRIDGE_BUILD F02_CORE_SHA="${CORE_SHA}")
set_target_properties(TrekF02Bridge PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(TrekF02Bridge PRIVATE /W4 /permissive- /EHsc)
endif()
enable_testing()
add_executable(f02_tests tests/f02_tests.cpp)
target_link_libraries(f02_tests PRIVATE TrekF02Bridge)
target_compile_features(f02_tests PRIVATE cxx_std_20)
foreach(case IN ITEMS identity placement invalid stale denied visibility context roundtrip rng corrupt incompatible scope thread buffers host_failure)
  add_test(NAME f02.${case} COMMAND f02_tests ${case})
endforeach()
