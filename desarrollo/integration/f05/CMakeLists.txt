cmake_minimum_required(VERSION 3.25)
project(TrekF05 LANGUAGES CXX)
set(F05_ENABLE_ORDERS ON CACHE BOOL "Explicit P01 integration" FORCE)
# F04 owns the one session, library and Core link. No second bridge/world.
add_subdirectory(../f04 "${CMAKE_CURRENT_BINARY_DIR}/f04")
# Reuse Core's audited JSON library/revision, without enabling its content backend.
include(FetchContent)
set(JSONCONS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(jsoncons SYSTEM GIT_REPOSITORY https://github.com/danielaparker/jsoncons.git GIT_TAG bcb44594c50c495ee1e690602cdd71455942ad0e)
FetchContent_MakeAvailable(jsoncons)
add_library(TrekF05Language STATIC src/f05_ai_json.cpp src/f05_ai_provider.cpp)
set_target_properties(TrekF05Language PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(TrekF05Language PUBLIC include ../f04/include)
target_compile_features(TrekF05Language PUBLIC cxx_std_20)
target_link_libraries(TrekF05Language PRIVATE jsoncons)
if(MSVC)
 target_compile_options(TrekF05Language PRIVATE /utf-8)
endif()
target_link_libraries(TrekF04Bridge PRIVATE TrekF05Language)
enable_testing()
add_executable(f05_tests tests/f05_tests.cpp)
target_link_libraries(f05_tests PRIVATE TrekF04Bridge CoreRPG::CoreRPG)
target_compile_features(f05_tests PRIVATE cxx_std_20)
# On Windows the test executable must share the existing bridge DLL directory.
set_target_properties(f05_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/f04")
foreach(case IN ITEMS scenario authority recipient unsupported scope stale reception
 duplicate revalidate expired cancel transition determinism corrupt buffers population history history_fault)
 add_test(NAME f05.${case} COMMAND f05_tests ${case} "${CMAKE_CURRENT_BINARY_DIR}/${case}")
endforeach()
add_test(NAME f05.process_save COMMAND f05_tests process_save "${CMAKE_CURRENT_BINARY_DIR}/orders")
set_tests_properties(f05.process_save PROPERTIES FIXTURES_SETUP orders)
add_test(NAME f05.process_load COMMAND f05_tests process_load "${CMAKE_CURRENT_BINARY_DIR}/orders")
set_tests_properties(f05.process_load PROPERTIES FIXTURES_REQUIRED orders)

add_executable(f05_delegation_tests tests/f05_delegation_tests.cpp)
target_link_libraries(f05_delegation_tests PRIVATE TrekF04Bridge CoreRPG::CoreRPG)
target_compile_features(f05_delegation_tests PRIVATE cxx_std_20)
set_target_properties(f05_delegation_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/f04")
foreach(case IN ITEMS valid scope authority revoke suspend expiry relief origin medical subdelegate modes mode_change idempotency causal history population restore_and_access roster_relief)
 add_test(NAME f05.p02_${case} COMMAND f05_delegation_tests ${case} "${CMAKE_CURRENT_BINARY_DIR}/p02-${case}")
endforeach()
add_test(NAME f05.p02_process_save COMMAND f05_delegation_tests process_save "${CMAKE_CURRENT_BINARY_DIR}/p02-process")
set_tests_properties(f05.p02_process_save PROPERTIES FIXTURES_SETUP delegation)
add_test(NAME f05.p02_process_load COMMAND f05_delegation_tests process_load "${CMAKE_CURRENT_BINARY_DIR}/p02-process")
set_tests_properties(f05.p02_process_load PROPERTIES FIXTURES_REQUIRED delegation)

add_executable(f05_initiative_tests tests/f05_initiative_tests.cpp)
target_link_libraries(f05_initiative_tests PRIVATE TrekF04Bridge CoreRPG::CoreRPG)
target_compile_features(f05_initiative_tests PRIVATE cxx_std_20)
set_target_properties(f05_initiative_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/f04")
foreach(case IN ITEMS scenario authority delegation revoked relief medical watch_end player player_captain knowledge bounded determinism idempotency corrupt history history_fault population population_relief preissue_relief selection requirement communication scope atomic)
 add_test(NAME f05.p03_${case} COMMAND f05_initiative_tests ${case} "${CMAKE_CURRENT_BINARY_DIR}/p03-${case}")
endforeach()
add_test(NAME f05.p03_process_save COMMAND f05_initiative_tests process_save "${CMAKE_CURRENT_BINARY_DIR}/p03-process")
set_tests_properties(f05.p03_process_save PROPERTIES FIXTURES_SETUP initiative)
add_test(NAME f05.p03_process_load COMMAND f05_initiative_tests process_load "${CMAKE_CURRENT_BINARY_DIR}/p03-process")

add_executable(f05_ai_tests tests/f05_ai_tests.cpp)
target_link_libraries(f05_ai_tests PRIVATE TrekF04Bridge TrekF05Language CoreRPG::CoreRPG jsoncons)
if(MSVC)
 target_compile_options(f05_ai_tests PRIVATE /utf-8)
endif()
set_target_properties(f05_ai_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/f04")
foreach(case IN ITEMS valid ambiguous insufficient advice privacy hidden binding malformed parameters injection authority delegation revoked approval player stale late out_of_order accept_stale offline timeout quota http network empty deterministic pending duplicate recovery history corrupt scope provider_valid provider_malformed provider_exception routing)

add_executable(f05_integration_audit_tests tests/f05_integration_audit_tests.cpp)
target_link_libraries(f05_integration_audit_tests PRIVATE TrekF04Bridge TrekF05Language CoreRPG::CoreRPG jsoncons)
set_target_properties(f05_integration_audit_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/f04")
foreach(case IN ITEMS A_captain B_command B_cancel_successor C_revoked C_suspended C_expired C_mode
 D_player E_player_captain F_officers F_contradictions G_timeout G_quota G_corrupt G_injection G_exception
 H_offscreen J_time limits capacity_ai capacity_orders capacity_delegations capacity_decisions

 add_test(NAME f05.p05a_${case} COMMAND f05_integration_audit_tests ${case} "${CMAKE_CURRENT_BINARY_DIR}/p05a-${case}")
endforeach()
add_test(NAME f05.p05a_I_process_save COMMAND f05_integration_audit_tests I_process_save "${CMAKE_CURRENT_BINARY_DIR}/p05a-process")
