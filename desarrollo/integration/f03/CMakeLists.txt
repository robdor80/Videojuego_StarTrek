cmake_minimum_required(VERSION 3.25)
project(TrekF03 LANGUAGES CXX)
set(CORERPG_SOURCE_DIR "" CACHE PATH "Audited CoreRPG F03 worktree")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/core-revision.txt" EXPECTED_CORE_SHA)
string(STRIP "${EXPECTED_CORE_SHA}" EXPECTED_CORE_SHA)
execute_process(COMMAND git -C "${CORERPG_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE CORE_SHA OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT CORE_SHA STREQUAL EXPECTED_CORE_SHA)
  message(FATAL_ERROR "Unaudited CoreRPG revision ${CORE_SHA}, expected ${EXPECTED_CORE_SHA}")
endif()
execute_process(COMMAND git -C "${CORERPG_SOURCE_DIR}" diff --exit-code HEAD RESULT_VARIABLE CORE_DIRTY OUTPUT_QUIET)
if(NOT CORE_DIRTY EQUAL 0)
  message(FATAL_ERROR "CoreRPG tracked sources differ from audited commit")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Core suite is run separately" FORCE)
set(CORERPG_BUILD_CONTENT_JSON OFF CACHE BOOL "Declarative in-memory fixture catalog" FORCE)
add_subdirectory("${CORERPG_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/corerpg")
set_target_properties(CoreRPG PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(TrekF03Bridge SHARED src/f03_bridge.cpp)
target_link_libraries(TrekF03Bridge PRIVATE CoreRPG::CoreRPG)
target_include_directories(TrekF03Bridge PUBLIC include)
target_compile_features(TrekF03Bridge PRIVATE cxx_std_20)
target_compile_definitions(TrekF03Bridge PRIVATE F03_BRIDGE_BUILD F03_CORE_SHA="${CORE_SHA}")
if(MSVC)
  target_compile_options(TrekF03Bridge PRIVATE /W4 /permissive- /EHsc)
endif()
enable_testing()
add_executable(f03_tests tests/f03_tests.cpp)
target_link_libraries(f03_tests PRIVATE TrekF03Bridge CoreRPG::CoreRPG)
target_compile_features(f03_tests PRIVATE cxx_std_20)
foreach(case IN ITEMS identity determinism observers hidden acquire report denied readonly offscreen order bounded duplicate backwards roundtrip rng corrupt incompatible scope thread buffers stale calendar tampered)
  add_test(NAME f03.${case} COMMAND f03_tests ${case})
endforeach()
add_test(NAME f03.process_save COMMAND f03_tests process_save "${CMAKE_CURRENT_BINARY_DIR}/process-campaign.f03")
set_tests_properties(f03.process_save PROPERTIES FIXTURES_SETUP campaign_file)
add_test(NAME f03.process_load COMMAND f03_tests process_load "${CMAKE_CURRENT_BINARY_DIR}/process-campaign.f03")
set_tests_properties(f03.process_load PROPERTIES FIXTURES_REQUIRED campaign_file)
